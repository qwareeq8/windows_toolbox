```python
def test_restore_maximized_window():
    """Restore of a previously-maximized window issues SW_MAXIMIZE (SNAP-04)."""
    import ctypes
    from ctypes import wintypes

    import win32con
    import win32gui

    from virelo.services.snap import ShiftSnapRestore

    # Create ShiftSnapRestore bypassing __init__ (avoids keyboard hooks and EnumWindows)
    mgr = ShiftSnapRestore.__new__(ShiftSnapRestore)
    mgr.settings = MockSettings()

    test_hwnd = 12345

    # Simulate a window that was maximized before snap
    mgr._orig_sizes = {
        test_hwnd: {
            "rect": (100, 100, 800, 600),
            "maximized": True,
        }
    }

    # Patch the dependencies _restore needs:
    # 1. USER32.GetForegroundWindow -- not called by _restore directly
    # 2. USER32.GetWindowRect -- fills a RECT struct, needs to NOT cover the full monitor
    #    (otherwise the fullscreen early-return on lines 347-353 would trigger)
    # 3. get_monitor_rect -- returns monitor bounds
    # 4. win32gui.ShowWindow -- the assertion target

    def fake_get_window_rect(hwnd, rect_ptr):
        """Populate rect with a small window (not covering monitor)."""
        rect = ctypes.cast(rect_ptr, ctypes.POINTER(wintypes.RECT)).contents
        rect.left = 200
        rect.top = 200
        rect.right = 600
        rect.bottom = 500

    with patch("virelo.services.snap.get_monitor_rect", return_value=(0, 0, 1920, 1080)):
        # Patch USER32.GetWindowRect to return a small window
        original_get_rect = mgr.__class__.__module__
        with patch("virelo.services.snap.USER32") as mock_user32:
            mock_user32.GetWindowRect.side_effect = fake_get_window_rect

            # Reset ShowWindow mock to track calls cleanly
            win32gui.ShowWindow.reset_mock()

            mgr._restore(test_hwnd)

    # The key assertion: ShowWindow called with SW_MAXIMIZE because was_maximized=True
    win32gui.ShowWindow.assert_called_once_with(test_hwnd, win32con.SW_MAXIMIZE)

    # Verify the hwnd was removed from _orig_sizes (consumed by restore)
    assert test_hwnd not in mgr._orig_sizes
```

Key points:
- Uses `ShiftSnapRestore.__new__` to bypass `__init__` (avoids keyboard hooks and win32gui.EnumWindows calls that need mocking). This pattern is established in PATTERNS.md.
- The conftest stubs provide `win32gui.ShowWindow` as a `MagicMock()`, so `assert_called_once_with` works.
- `get_monitor_rect` is patched to return a valid monitor rect `(0, 0, 1920, 1080)`.
- `USER32.GetWindowRect` is patched via side_effect to populate the RECT struct with a small window (200, 200, 600, 500) that does NOT cover the full monitor -- this avoids the fullscreen early-return guard.
- The `_restore` method pops `test_hwnd` from `_orig_sizes`, checks `was_maximized=True`, and calls `win32gui.ShowWindow(hwnd, win32con.SW_MAXIMIZE)`.
